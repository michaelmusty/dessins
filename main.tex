\begin{frame}[plain]{Belyi maps, dessins, etc.}
    % some definitions
    \begin{vfilleditems}
        \item This branched covering is an example of a \textbf{Belyi map},
        a map of Riemann surfaces unbranched outside of $0,1,\infty$.
        \item A Belyi map determines a \textbf{permutation triple} via monodromy.
        \item A permutation triple determines a \textbf{triangle subgroup} $\Gamma$
        which acts on a spherical, Euclidean, or hyperbolic space $\mathcal{H}$
        such that $\Gamma \backslash \mathcal{H} \to \Delta \backslash \mathcal{H}$
        is the corresponding Belyi map.
        \item The preimage of $[0,1]$ is a (connected) bipartite graph
        embedded on a surface called a \textbf{dessin d'enfant}.
    \end{vfilleditems}
\end{frame}

\begin{frame}[plain]{Galois action}
    \begin{theorem}[Belyi, 1979]
        A smooth projective curve $X$ over $\C$
        can be defined over $\overline{\Q}$ if and only if there
        exists a nonconstant morphism of algebraic curves $\varphi:X\to\P^1$
        unramified outside $\{0,1,\infty\}$.
      \end{theorem}
      \vfill
      In the 1980s Grothendieck described an action of the absolute Galois group
      $\Gal(\overline{\Q}/\Q)$ on the set of (isomorphism classes of) dessins.

\end{frame}
\begin{frame}[plain]{Passports}
    % passport, genus, galois orbit, size

A \textbf{passport} consists of the data $(G,\lambda)$ where
\begin{vfilleditems}
\item
$G \leq S_d$ is a transitive subgroup
\item
$\lambda=(\lambda_0,\lambda_1,\lambda_\infty)$ is a triple of partitions of $d$
\end{vfilleditems}
A permutation triple $\sigma \in S_d^3$ \textbf{belongs} to $(G, \lambda)$ if
\begin{vfilleditems}
\item
$\langle \sigma_0, \sigma_1, \sigma_\infty \rangle = G$
\item
$\sigma_0, \sigma_1, \sigma_\infty$ have the cycle types specified by $\lambda_0, \lambda_1, \lambda_\infty$
\end{vfilleditems}
Note
\begin{vfilleditems}
\item The \textbf{size} of a passport is the number of permutation triples belonging to it (up to simultaneous conjugacy).

\item A \textbf{refined passport} replaces $\lambda$ with $C=(C_0,C_1,C_\infty)$
a triple of conjugacy classes of $G$.

\item Passports and refined passports are stable under the Galois action and therefore are
unions of Galois orbits.
\end{vfilleditems}

\end{frame}

\begin{frame}[plain]{Motivating future questions}
%   one thing that you might want to say, even if you don't write it, is that the goal is to learn something about the absolute Galois group through its action on dessins
% you could also add a slide with some motivating future questions
% like, can we tell when 2 dessins correspond to Galois conjugate Belyi maps
% is there topological or combinatorial data that can tell us when a passport splits into multiple Galois orbits
The ultimate goal is to learn something about the absolute Galois group through its action on dessins.
To that end:
\begin{vfilleditems}
  \item Can we tell when 2 dessins correspond to Galois conjugate Belyi maps?
  \item Is there topological or combinatorial data that can tell us when a passport splits into multiple Galois orbits?
  \item Can we determine the Galois orbits given a set of representatives for a passport?
  \item For every passport in the LMFDB with multiple Galois orbits, can we give an explicit reason for that splitting?
\end{vfilleditems}

\end{frame}
