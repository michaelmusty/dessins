\section{motivating example}
\begin{frame}{sales rep assignment}
    \begin{vfilleditems}
        \pause
        \item A solar company Acme Inc. has inbound residential customers (leads $l_i$)
        \pause
        \item Each lead schedules a day to have an Acme sales rep $r_j$ visit the lead's residence
        \pause
        \item Acme has the ability to determine the expected profit $e_{ij}$ from having $r_j$ visit $l_i$
        \pause
        \item \textbf{objective}: Determine which rep to assign to each lead to maximize the sum of expected profits each day
        \pause
        \item subject to the \textbf{constraints}:
        \begin{vfilleditems}
            \item every lead get assigned exactly 1 rep
            \item every rep visits at most 1 lead per day
        \end{vfilleditems}
    \end{vfilleditems}
\end{frame}
\begin{frame}{setup}
    \begin{align*}
        l_i &= \text{ lead $i$},\quad i\in\{0,\dots,L-1\}\\
        r_j &= \text{ rep $j$},\quad j\in\{0,\dots,R-1\}\\
        e_{ij} &= \text{ expected profit of assigning $l_i$ to $r_j$}\\
        x_{ij} &= \begin{cases}
            1 &\text{ if $l_i$ is assigned to $r_j$}\\
            0 &\text{ otherwise}
        \end{cases}
    \end{align*}
\end{frame}
\begin{frame}{objective and constraints}
    \pause
    \begin{vfilleditems}
        \item
        \textbf{objective}: choose $x_{ij}$ to maximize
        \begin{align*}
            \sum_{i,j}e_{ij}x_{ij}
        \end{align*}
        \pause
        \item subject to the
        \textbf{constraints}:
        \begin{vfilleditems}
            \item $\sum_{j=0}^{R-1} x_{ij} = 1$, for every $l_i$
            \pause
            \newline
            i.e. \emph{every lead gets assigned exactly 1 rep}
            \pause
            \item $\sum_{i=0}^{L-1} x_{ij} \leq 1$, for every $r_j$
            \pause
            \newline
            i.e. \emph{every rep visits at most 1 lead per day}
        \end{vfilleditems}
    \end{vfilleditems}
\end{frame}
\begin{frame}{example $L=2$, $R=3$}
    \begin{vfilleditems}
        \pause
        \item Given expected profits
        \begin{align*}
            (e_{ij}) &=\begin{pmatrix}
                4&5&2\\
                1&3&1
            \end{pmatrix},
            \quad\text{i.e. } e_{11}=3, e_{01}=5,\dots
        \end{align*}
        \pause
        \item
        \only<3>{
            maximize
            \begin{align*}
                \sum_{i,j}e_{ij}x_{ij} &= e_{00}x_{00}+e_{01}x_{01}+e_{02}x_{02}\\
                &+e_{10}x_{10}+e_{11}x_{11}+e_{12}x_{12}
            \end{align*}
        }
        \only<4>{
            maximize
            \begin{align*}
                \sum_{i,j}e_{ij}x_{ij} &= 4\cdot x_{00}+5\cdot x_{01}+2\cdot x_{02}\\
                &+1\cdot x_{10}+3\cdot x_{11}+1\cdot x_{12}
            \end{align*}
        }
   \end{vfilleditems}
\end{frame}

\begin{frame}{example $L=2, R=3$}
\begin{vfilleditems}
    \item
    subject to the constraints
    \pause
    \item every lead gets assigned exactly 1 rep
        \begin{align*}
            x_{00}+x_{01}+x_{02}&=1, (l_0)\\
            x_{10}+x_{11}+x_{12}&=1, (l_1)
        \end{align*}
    \pause
    \item every rep visits at most 1 lead

\end{vfilleditems}
\end{frame}

\begin{frame}{combinatorial optimization}
    \begin{vfilleditems}
        \pause
        \item difficult in general but can be done in practice
        \pause
        \item\textbf{integer linear programming}:
        \begin{vfilleditems}
            \item linear objective and constraints
            \item integer decision variables
        \end{vfilleditems}
        \pause
        \item\textbf{constraint programming}:
        \begin{vfilleditems}
            \item determine feasible solutions to a system of constraints
            \item can be used to solve integer linear programs
        \end{vfilleditems}
    \end{vfilleditems}
\end{frame}

\begin{frame}{setup with time slots}
    Now suppose that leads sign up for one of $T$ daily time slots and that rep availability can vary
    \pause
    \begin{align*}
        l_i &= \text{ lead $i$},\quad i\in\{0,\dots,L-1\}\\
        r_j &= \text{ rep $j$},\quad j\in\{0,\dots,R-1\}\\
        t_k &= \text{ time slot $k$}, \quad k\in\{0,\dots,T-1\}\\
        t_{s_i} &= \text{ time slot for $l_i$},\quad s_i\in\{0,\dots,T-1\}\\
        e_{ij} &= \text{ expected profit of assigning $l_i$ to $r_j$}\\
        x_{ij} &= \begin{cases}
            1 &\text{ if $l_i$ is assigned to $r_j$}\\
            0 &\text{ otherwise}
        \end{cases}\\
        a_{jk} &= \begin{cases}
            1&\text{ if $r_j$ is available at $t_k$}\\
            0&\text{ otherwise}
        \end{cases}
    \end{align*}
\end{frame}
\begin{frame}{objective}
We aim to maximize
\begin{align*}
    \sum_{i,j}e_{ij}x_{ij}
\end{align*}
subject to the following constraints
\end{frame}
\begin{frame}{constraints}
\begin{vfilleditems}
    \item For every $l_i$
    \begin{align*}
         \sum_{j=0}^{R-1} x_{ij}a_{js_i}= 1
    \end{align*}
        \pause
\item
Note that $a_{js_i}$ is the availability of $r_j$ at the time slot corresponding to $l_i$ (since this time slot has index $s_i$)
    \pause
    \item i.e. \emph{every lead must be assigned exactly 1 available rep}
\end{vfilleditems}
\end{frame}
\begin{frame}{constraints}
\begin{vfilleditems}
\item
For every $r_j$ and every $t_k$
\begin{align*}
    \sum_{i=0}^{L-1}x_{ij}a_{jk}\leq 1
\end{align*}
\pause
\item
i.e. \emph{a rep cannot have more than 1 assignment in each time slot}
\pause
\item Note that for a given $j$ we do allow
\begin{align*}
\sum_{i=0}^{L-1} x_{ij}a_{j s_i} \geq 1
\end{align*}
\end{vfilleditems}
\end{frame}
